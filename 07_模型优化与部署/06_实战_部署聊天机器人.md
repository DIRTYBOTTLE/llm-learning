## 二、使用 Gradio 部署聊天机器人

### 完整代码

```python
import gradio as gr
from transformers import AutoModelForCausalLM, AutoTokenizer
import torch

class ChatBot:
    def __init__(self, model_name="TinyLlama/TinyLlama-1.1B-Chat-v1.0"):
        """初始化聊天机器人"""
        
        print(f"正在加载模型：{model_name}")
        
        self.tokenizer = AutoTokenizer.from_pretrained(model_name)
        self.model = AutoModelForCausalLM.from_pretrained(
            model_name,
            torch_dtype=torch.float16,
            device_map="auto"
        )
        
        # 设置 pad_token
        if self.tokenizer.pad_token is None:
            self.tokenizer.pad_token = self.tokenizer.eos_token
        
        print("模型加载完成！")
    
    def generate_response(self, message, history, max_length=500, temperature=0.7):
        """生成回复"""
        
        # 构造对话历史
        prompt = ""
        for user_msg, assistant_msg in history:
            prompt += f"User: {user_msg}\nAssistant: {assistant_msg}\n"
        prompt += f"User: {message}\nAssistant: "
        
        # 编码
        input_ids = self.tokenizer(prompt, return_tensors="pt").to(self.model.device)
        
        # 生成
        with torch.no_grad():
            output = self.model.generate(
                **input_ids,
                max_length=max_length,
                temperature=temperature,
                do_sample=True,
                top_p=0.9,
                pad_token_id=self.tokenizer.eos_token_id
            )
        
        # 解码
        generated = self.tokenizer.decode(output[0], skip_special_tokens=True)
        
        # 提取助手回复
        response = generated.split("Assistant: ")[-1].strip()
        
        # 如果回复包含下一个 "User:"，只取之前的部分
        if "User:" in response:
            response = response.split("User:")[0].strip()
        
        return response

# 创建聊天机器人
chatbot = ChatBot()

# 创建 Gradio 界面
def respond(message, history):
    response = chatbot.generate_response(message, history)
    return response

demo = gr.ChatInterface(
    fn=respond,
    title="AI 聊天机器人",
    description="这是一个基于 TinyLlama 的聊天机器人",
    examples=[
        "你好，你是谁？",
        "请介绍一下人工智能。",
        "写一首关于春天的诗。"
    ],
    retry_btn="重新生成",
    undo_btn="撤销",
    clear_btn="清空"
)

# 启动
demo.launch(server_name="0.0.0.0", server_port=7860)
```

## 三、使用 vLLM 部署高性能 API

### 安装 vLLM

```bash
pip install vllm
```

### 启动 vLLM 服务

```bash
# 启动服务（兼容 OpenAI API）
python -m vllm.entrypoints.openai.api_server \
    --model TinyLlama/TinyLlama-1.1B-Chat-v1.0 \
    --host 0.0.0.0 \
    --port 8000
```

### 使用 API

```python
import requests
import json

def chat_with_bot(message, history=[]):
    """与聊天机器人对话"""
    
    # 构造消息
    messages = []
    for user_msg, assistant_msg in history:
        messages.append({"role": "user", "content": user_msg})
        messages.append({"role": "assistant", "content": assistant_msg})
    messages.append({"role": "user", "content": message})
    
    # 调用 API
    response = requests.post(
        "http://localhost:8000/v1/chat/completions",
        json={
            "model": "TinyLlama/TinyLlama-1.1B-Chat-v1.0",
            "messages": messages,
            "temperature": 0.7,
            "max_tokens": 500
        }
    )
    
    # 解析响应
    result = response.json()
    assistant_message = result["choices"][0]["message"]["content"]
    
    return assistant_message

# 测试
response = chat_with_bot("你好，你是谁？")
print(response)
```

## 四、使用 Ollama 部署（最简单）

### 安装 Ollama

```bash
# macOS/Linux
curl -fsSL https://ollama.com/install.sh | sh

# Windows
# 从 https://ollama.com 下载安装
```

### 拉取并运行模型

```bash
# 拉取模型
ollama pull llama2

# 运行对话
ollama run llama2

# 启动 API 服务
ollama serve
```

### 使用 Ollama API

```python
import requests

def chat_with_ollama(message, model="llama2"):
    """使用 Ollama API 对话"""
    
    response = requests.post(
        "http://localhost:11434/api/generate",
        json={
            "model": model,
            "prompt": message,
            "stream": False
        }
    )
    
    return response.json()["response"]

# 测试
response = chat_with_ollama("你好，你是谁？")
print(response)
```


### 使用 Hugging Face Spaces

```python
# app.py
import gradio as gr
from transformers import pipeline

# 使用 pipeline（更简单）
generator = pipeline("text-generation", model="TinyLlama/TinyLlama-1.1B-Chat-v1.0")

def respond(message, history):
    prompt = ""
    for user_msg, assistant_msg in history:
        prompt += f"User: {user_msg}\nAssistant: {assistant_msg}\n"
    prompt += f"User: {message}\nAssistant: "
    
    result = generator(prompt, max_length=500, temperature=0.7)
    response = result[0]["generated_text"].split("Assistant: ")[-1].strip()
    
    return response

demo = gr.ChatInterface(fn=respond, title="AI Chatbot")
demo.launch()
```

## 七、总结

### 部署方式对比

| 方式 | 难度 | 性能 | 适用场景 |
|------|------|------|----------|
| Gradio | 简单 | 中等 | 演示、原型 |
| vLLM | 中等 | 高 | 生产环境 |
| Ollama | 最简单 | 中等 | 本地使用 |

### 部署建议

```
快速演示：Gradio
生产环境：vLLM 或 TGI
本地使用：Ollama
边缘设备：llama.cpp
```
