## 二、API 部署

### 使用 FastAPI 部署

```python
from fastapi import FastAPI
from transformers import AutoModelForCausalLM, AutoTokenizer
import torch

app = FastAPI()

# 加载模型
model_name = "meta-llama/Llama-2-7b-hf"
tokenizer = AutoTokenizer.from_pretrained(model_name)
model = AutoModelForCausalLM.from_pretrained(
    model_name,
    torch_dtype=torch.float16,
    device_map="auto"
)

@app.post("/generate")
async def generate(request: dict):
    """生成文本"""
    
    prompt = request.get("prompt", "")
    max_length = request.get("max_length", 100)
    temperature = request.get("temperature", 0.7)
    
    # 编码
    input_ids = tokenizer(prompt, return_tensors="pt").to(model.device)
    
    # 生成
    with torch.no_grad():
        output = model.generate(
            **input_ids,
            max_length=max_length,
            temperature=temperature,
            do_sample=True
        )
    
    # 解码
    generated_text = tokenizer.decode(output[0], skip_special_tokens=True)
    
    return {"generated_text": generated_text}

# 启动服务
# uvicorn app:app --host 0.0.0.0 --port 8000
```

### 使用 vLLM 部署（兼容 OpenAI API）

```python
from vllm import AsyncLLMEngine, SamplingParams
from fastapi import FastAPI
from pydantic import BaseModel

app = FastAPI()

# 加载模型
engine = AsyncLLMEngine.from_pretrained("meta-llama/Llama-2-7b-hf")

class ChatRequest(BaseModel):
    messages: list
    temperature: float = 0.7
    max_tokens: int = 100

@app.post("/v1/chat/completions")
async def chat(request: ChatRequest):
    """兼容 OpenAI 的聊天 API"""
    
    # 构造 prompt
    prompt = ""
    for msg in request.messages:
        prompt += f"{msg['role']}: {msg['content']}\n"
    prompt += "assistant: "
    
    # 生成
    sampling_params = SamplingParams(
        temperature=request.temperature,
        max_tokens=request.max_tokens
    )
    
    outputs = await engine.generate(prompt, sampling_params)
    generated = outputs[0].outputs[0].text
    
    return {
        "choices": [{
            "message": {
                "role": "assistant",
                "content": generated
            }
        }]
    }
```

## 三、Web 部署

### 使用 Gradio

```python
import gradio as gr
from transformers import AutoModelForCausalLM, AutoTokenizer
import torch

# 加载模型
model_name = "meta-llama/Llama-2-7b-hf"
tokenizer = AutoTokenizer.from_pretrained(model_name)
model = AutoModelForCausalLM.from_pretrained(
    model_name,
    torch_dtype=torch.float16,
    device_map="auto"
)

def generate_text(prompt, max_length, temperature):
    """生成文本"""
    
    input_ids = tokenizer(prompt, return_tensors="pt").to(model.device)
    
    with torch.no_grad():
        output = model.generate(
            **input_ids,
            max_length=max_length,
            temperature=temperature,
            do_sample=True
        )
    
    generated = tokenizer.decode(output[0], skip_special_tokens=True)
    return generated

# 创建界面
interface = gr.Interface(
    fn=generate_text,
    inputs=[
        gr.Textbox(label="输入提示"),
        gr.Slider(minimum=10, maximum=500, value=100, label="最大长度"),
        gr.Slider(minimum=0.1, maximum=2.0, value=0.7, label="温度")
    ],
    outputs=gr.Textbox(label="生成结果"),
    title="AI 文本生成",
    description="输入提示，AI 会生成续写"
)

# 启动
interface.launch(server_name="0.0.0.0", server_port=7860)
```

### 使用 Streamlit

```python
import streamlit as st
from transformers import AutoModelForCausalLM, AutoTokenizer
import torch

# 加载模型
@st.cache_resource
def load_model():
    model_name = "meta-llama/Llama-2-7b-hf"
    tokenizer = AutoTokenizer.from_pretrained(model_name)
    model = AutoModelForCausalLM.from_pretrained(
        model_name,
        torch_dtype=torch.float16,
        device_map="auto"
    )
    return model, tokenizer

model, tokenizer = load_model()

# 界面
st.title("AI 文本生成")

prompt = st.text_area("输入提示", "Today the weather is")
max_length = st.slider("最大长度", 10, 500, 100)
temperature = st.slider("温度", 0.1, 2.0, 0.7)

if st.button("生成"):
    input_ids = tokenizer(prompt, return_tensors="pt").to(model.device)
    
    with torch.no_grad():
        output = model.generate(
            **input_ids,
            max_length=max_length,
            temperature=temperature,
            do_sample=True
        )
    
    generated = tokenizer.decode(output[0], skip_special_tokens=True)
    st.write("生成结果：")
    st.write(generated)
```

## 四、边缘部署

### 使用 llama.cpp 部署到电脑

```bash
# 1. 下载 GGUF 模型
wget https://huggingface.co/TheBloke/Llama-2-7B-GGUF/resolve/main/llama-2-7b.Q4_0.gguf

# 2. 运行推理
./main -m llama-2-7b.Q4_0.gguf -p "Hello" -n 100

# 3. 启动 API 服务器
./server -m llama-2-7b.Q4_0.gguf --host 0.0.0.0 --port 8080
```

### 使用 Ollama 部署

```bash
# 安装 Ollama
curl -fsSL https://ollama.com/install.sh | sh

# 拉取模型
ollama pull llama2

# 运行
ollama run llama2

# 启动 API 服务
ollama serve
```

## 五、部署最佳实践

### 选择部署方式

```
┌─────────────────────────────────────────────────────────────┐
│                    如何选择部署方式？                        │
├─────────────────────────────────────────────────────────────┤
│                                                             │
│  高并发 API 服务：                                          │
│    → vLLM 或 TGI                                            │
│                                                             │
│  快速原型展示：                                              │
│    → Gradio 或 Streamlit                                    │
│                                                             │
│  私有化部署：                                                │
│    → llama.cpp 或 Ollama                                    │
│                                                             │
│  边缘设备：                                                  │
│    → llama.cpp + INT4 量化                                   │
│                                                             │
└─────────────────────────────────────────────────────────────┘
```

### 性能优化

```
1. 使用量化模型
   减少显存占用，提高速度
   
2. 使用批处理
   提高吞吐量
   
3. 使用缓存
   KV Cache、结果缓存
   
4. 使用专用框架
   vLLM、TGI 等
```

## 六、总结

### 部署方式对比

| 方式 | 优点 | 缺点 | 适用场景 |
|------|------|------|----------|
| API | 灵活、可扩展 | 需要服务器 | 后端服务 |
| Web | 用户友好 | 需要前端 | 直接面向用户 |
| 边缘 | 离线、隐私 | 性能有限 | 私有化部署 |

### 常用工具

| 工具 | 用途 |
|------|------|
| FastAPI | API 服务 |
| Gradio | 快速 Web 界面 |
| vLLM | 高性能推理 |
| TGI | Hugging Face 推理服务 |
| llama.cpp | CPU/边缘推理 |
| Ollama | 简单部署 |
