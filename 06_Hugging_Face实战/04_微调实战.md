### 数据预处理

```python
def preprocess_data(data, tokenizer):
    """预处理训练数据"""
    
    processed_data = []
    
    for sample in data:
        # 构造对话文本
        conversation = ""
        for turn in sample["conversations"]:
            if turn["role"] == "user":
                conversation += f"用户：{turn['content']} 助手："
            else:
                conversation += turn['content']
        
        # 分词
        encodings = tokenizer(conversation, truncation=True, max_length=512)
        
        # 构造标签（只计算助手回答部分的损失）
        input_ids = encodings["input_ids"]
        labels = [-100] * len(input_ids)  # 先全部设为 -100
        
        # 找到助手回答的位置，设置正确的标签
        # （简化处理，实际需要更复杂的逻辑）
        
        processed_data.append({
            "input_ids": input_ids,
            "attention_mask": encodings["attention_mask"],
            "labels": labels
        })
    
    return processed_data
```

## 三、使用 Trainer 训练

### 完整的训练代码

```python
from transformers import (
    AutoModelForCausalLM,
    AutoTokenizer,
    TrainingArguments,
    Trainer,
    DataCollatorForLanguageModeling
)
from datasets import Dataset
import torch

def train_model():
    """使用 Trainer 训练模型"""
    
    # 1. 加载模型和分词器
    model_name = "gpt2"
    model = AutoModelForCausalLM.from_pretrained(model_name)
    tokenizer = AutoTokenizer.from_pretrained(model_name)
    
    # 设置 padding token
    tokenizer.pad_token = tokenizer.eos_token
    model.config.pad_token_id = tokenizer.eos_token_id
    
    # 2. 准备训练数据
    train_texts = [
        "用户：你好 助手：你好！有什么可以帮你的吗？",
        "用户：1+1等于几？ 助手：1+1等于2。",
        "用户：请介绍一下北京。 助手：北京是中国的首都。",
    ]
    
    # 分词
    def tokenize_function(examples):
        return tokenizer(examples["text"], truncation=True, max_length=128)
    
    train_dataset = Dataset.from_dict({"text": train_texts})
    train_dataset = train_dataset.map(tokenize_function, batched=True)
    
    # 3. 配置训练参数
    training_args = TrainingArguments(
        output_dir="./output",
        num_train_epochs=3,
        per_device_train_batch_size=2,
        learning_rate=5e-5,
        warmup_steps=10,
        logging_steps=10,
        save_steps=100,
        save_total_limit=2,
    )
    
    # 4. 创建 Trainer
    trainer = Trainer(
        model=model,
        args=training_args,
        train_dataset=train_dataset,
        data_collator=DataCollatorForLanguageModeling(tokenizer, mlm=False),
    )
    
    # 5. 开始训练
    trainer.train()
    
    # 6. 保存模型
    trainer.save_model("./fine-tuned-model")
    tokenizer.save_pretrained("./fine-tuned-model")
    
    return model, tokenizer

# 训练
model, tokenizer = train_model()
```

## 四、使用 LoRA 微调

### 安装依赖

```bash
pip install peft
```

### LoRA 微调代码

```python
from transformers import AutoModelForCausalLM, AutoTokenizer, TrainingArguments, Trainer
from peft import LoraConfig, get_peft_model, TaskType
import torch

def train_with_lora():
    """使用 LoRA 微调模型"""
    
    # 1. 加载模型
    model_name = "gpt2"
    model = AutoModelForCausalLM.from_pretrained(model_name)
    tokenizer = AutoTokenizer.from_pretrained(model_name)
    tokenizer.pad_token = tokenizer.eos_token
    
    # 2. 配置 LoRA
    lora_config = LoraConfig(
        task_type=TaskType.CAUSAL_LM,
        r=8,                      # 低秩矩阵的秩
        lora_alpha=32,            # 缩放因子
        lora_dropout=0.05,        # Dropout
        target_modules=["c_attn", "c_proj"],  # 要应用 LoRA 的模块
    )
    
    # 3. 应用 LoRA
    model = get_peft_model(model, lora_config)
    
    # 查看可训练参数
    model.print_trainable_parameters()
    # 输出：trainable params: 147,456 || all params: 124,587,264 || trainable%: 0.12
    
    # 4. 准备数据
    train_texts = [
        "用户：你好 助手：你好！有什么可以帮你的吗？",
        "用户：1+1等于几？ 助手：1+1等于2。",
    ]
    
    from datasets import Dataset
    def tokenize_function(examples):
        return tokenizer(examples["text"], truncation=True, max_length=128)
    
    train_dataset = Dataset.from_dict({"text": train_texts})
    train_dataset = train_dataset.map(tokenize_function, batched=True)
    
    # 5. 训练
    training_args = TrainingArguments(
        output_dir="./lora-output",
        num_train_epochs=3,
        per_device_train_batch_size=2,
        learning_rate=2e-4,  # LoRA 通常用较大学习率
        logging_steps=10,
    )
    
    trainer = Trainer(
        model=model,
        args=training_args,
        train_dataset=train_dataset,
    )
    
    trainer.train()
    
    # 6. 保存 LoRA 权重
    model.save_pretrained("./lora-weights")
    
    return model, tokenizer

# 训练
model, tokenizer = train_with_lora()
```

### QLoRA 微调代码

from transformers import AutoModelForCausalLM, AutoTokenizer, BitsAndBytesConfig, TrainingArguments, Trainer
from peft import LoraConfig, get_peft_model, prepare_model_for_kbit_training, TaskType
import torch

def train_with_qlora():
    """使用 QLoRA 微调模型"""
    
    # 1. 量化配置
    bnb_config = BitsAndBytesConfig(
        load_in_4bit=True,                  # 4-bit 量化
        bnb_4bit_quant_type="nf4",          # NF4 量化
        bnb_4bit_compute_dtype=torch.float16,
        bnb_4bit_use_double_quant=True,     # 双重量化
    )
    
    # 2. 加载量化模型
    model_name = "meta-llama/Llama-2-7b-hf"
    model = AutoModelForCausalLM.from_pretrained(
        model_name,
        quantization_config=bnb_config,
        device_map="auto",
    )
    tokenizer = AutoTokenizer.from_pretrained(model_name)
    tokenizer.pad_token = tokenizer.eos_token
    
    # 3. 准备模型用于 k-bit 训练
    model = prepare_model_for_kbit_training(model)
    
    # 4. 配置 LoRA
    lora_config = LoraConfig(
        task_type=TaskType.CAUSAL_LM,
        r=8,
        lora_alpha=32,
        target_modules=["q_proj", "v_proj"],
        lora_dropout=0.05,
        bias="none",
    )
    
    # 5. 应用 LoRA
    model = get_peft_model(model, lora_config)
    model.print_trainable_parameters()
    
    # 6. 准备数据（同上）
    # ...
    
    # 7. 训练

        output_dir="./qlora-output",
        num_train_epochs=3,
        per_device_train_batch_size=4,

## 六、测试微调效果

### 测试代码

```python
def test_model(model, tokenizer, prompt):
    """测试微调后的模型"""
    
    # 构造输入
    input_text = f"用户：{prompt} 助手："
    input_ids = tokenizer(input_text, return_tensors="pt")
    
    # 生成
    with torch.no_grad():
        output = model.generate(
            **input_ids,
            max_length=100,
            temperature=0.7,
            do_sample=True,
            pad_token_id=tokenizer.eos_token_id
        )
    
    # 解码
    generated_text = tokenizer.decode(output[0], skip_special_tokens=True)
    
    # 提取助手的回答
    response = generated_text.split("助手：")[-1].strip()
    
    return response

# 测试
prompts = ["你好", "1+1等于几？", "请介绍一下北京。"]

for prompt in prompts:
    response = test_model(model, tokenizer, prompt)
    print(f"用户：{prompt}")
    print(f"助手：{response}")
    print()
```
