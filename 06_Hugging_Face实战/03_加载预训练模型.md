## 一、AutoClass 简介

### 什么是 AutoClass？

```
AutoClass 是 Transformers 库中的自动类

作用：
  - 根据模型名称自动选择正确的类
  - 不需要记住每个模型对应的类名
  
例子：
  "gpt2" → GPT2LMHeadModel
  "bert-base-uncased" → BertForMaskedLM
  "meta-llama/Llama-2-7b-hf" → LlamaForCausalLM
  
AutoClass 会自动处理！
```

### 常用的 AutoClass

```
┌─────────────────────────────────────────────────────────────┐
│                    常用的 AutoClass                          │
├─────────────────────────────────────────────────────────────┤
│  类名                      用途                             │
├─────────────────────────────────────────────────────────────┤
│  AutoModel                 通用模型基类                     │
│  AutoModelForCausalLM      因果语言模型（GPT类）            │
│  AutoModelForMaskedLM      掩码语言模型（BERT类）           │
│  AutoModelForSeq2SeqLM     序列到序列模型（T5类）           │
│  AutoModelForSequenceClassification 文本分类               │
│  AutoModelForQuestionAnswering 问答                        │
│  AutoTokenizer             分词器                           │
└─────────────────────────────────────────────────────────────┘
```

## 三、使用模型生成文本

### 基本生成

```python
from transformers import AutoModelForCausalLM, AutoTokenizer

# 加载模型和分词器
model = AutoModelForCausalLM.from_pretrained("gpt2")
tokenizer = AutoTokenizer.from_pretrained("gpt2")

# 输入文本
text = "Today the weather is"

# 编码
input_ids = tokenizer.encode(text, return_tensors="pt")

# 生成
output = model.generate(input_ids, max_length=50)

# 解码
generated_text = tokenizer.decode(output[0], skip_special_tokens=True)
print(generated_text)
```

### 设置生成参数

```python
from transformers import AutoModelForCausalLM, AutoTokenizer

model = AutoModelForCausalLM.from_pretrained("gpt2")
tokenizer = AutoTokenizer.from_pretrained("gpt2")

text = "Today the weather is"
input_ids = tokenizer.encode(text, return_tensors="pt")

# 设置参数生成
output = model.generate(
    input_ids,
    max_length=50,          # 最大长度
    num_return_sequences=3, # 生成3个结果
    temperature=0.7,        # 温度
    do_sample=True,         # 采样
    top_k=50,               # Top-K
    top_p=0.9,              # Top-P
    pad_token_id=tokenizer.eos_token_id
)

# 解码所有结果
for i, out in enumerate(output):
    generated = tokenizer.decode(out, skip_special_tokens=True)
    print(f"结果{i+1}: {generated}")
```

## 四、前向传播详解

### 手动前向传播

```python
from transformers import AutoModelForCausalLM, AutoTokenizer
import torch

model = AutoModelForCausalLM.from_pretrained("gpt2")
tokenizer = AutoTokenizer.from_pretrained("gpt2")

text = "Hello"
input_ids = tokenizer.encode(text, return_tensors="pt")

# 前向传播
with torch.no_grad():
    outputs = model(input_ids)

# 查看输出
print(f"输出类型: {type(outputs)}")
print(f"logits 形状: {outputs.logits.shape}")
# logits 形状: (1, 1, 50257)  → (batch_size, seq_len, vocab_size)

# 获取下一个 token 的概率
next_token_logits = outputs.logits[:, -1, :]  # 取最后一个位置
next_token_probs = torch.softmax(next_token_logits, dim=-1)

# 选择概率最大的 token
next_token_id = torch.argmax(next_token_probs, dim=-1)
next_token = tokenizer.decode(next_token_id)

print(f"预测的下一个词: {next_token}")
```

### 逐步生成

```python
from transformers import AutoModelForCausalLM, AutoTokenizer
import torch

model = AutoModelForCausalLM.from_pretrained("gpt2")
tokenizer = AutoTokenizer.from_pretrained("gpt2")

def generate_step_by_step(model, tokenizer, prompt, max_length=20):
    """逐步生成文本"""
    
    input_ids = tokenizer.encode(prompt, return_tensors="pt")
    
    print(f"初始输入: {prompt}")
    
    for i in range(max_length):
        # 前向传播
        with torch.no_grad():
            outputs = model(input_ids)
        
        # 获取下一个 token
        next_token_logits = outputs.logits[:, -1, :]
        next_token_id = torch.argmax(next_token_logits, dim=-1)
        
        # 添加到序列
        input_ids = torch.cat([input_ids, next_token_id.unsqueeze(0)], dim=-1)
        
        # 解码新 token
        new_token = tokenizer.decode(next_token_id)
        print(f"步骤{i+1}: 预测 '{new_token}'")
        
        # 如果是结束符，停止
        if next_token_id.item() == tokenizer.eos_token_id:
            break
    
    # 解码完整文本
    generated_text = tokenizer.decode(input_ids[0], skip_special_tokens=True)
    return generated_text

# 使用
result = generate_step_by_step(model, tokenizer, "Hello", max_length=10)
print(f"\n完整文本: {result}")
```

## 六、模型配置

### 查看模型配置

```python
from transformers import AutoConfig

# 加载配置
config = AutoConfig.from_pretrained("gpt2")

print(f"模型类型: {config.model_type}")
print(f"词汇表大小: {config.vocab_size}")
print(f"隐藏层维度: {config.n_embd}")
print(f"注意力头数: {config.n_head}")
print(f"层数: {config.n_layer}")
```

### 自定义配置

```python
from transformers import GPT2Config, GPT2LMHeadModel

# 自定义配置
config = GPT2Config(
    vocab_size=50257,
    n_positions=1024,
    n_embd=256,       # 较小的维度
    n_layer=4,         # 较少的层数
    n_head=4,          # 较少的注意力头
)

# 使用自定义配置创建模型
model = GPT2LMHeadModel(config)

# 查看参数量
total_params = sum(p.numel() for p in model.parameters())
print(f"总参数量: {total_params:,}")
```

## 七、使用 GPU

### 将模型移到 GPU

```python
import torch
from transformers import AutoModelForCausalLM, AutoTokenizer

# 检查 GPU
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
print(f"使用设备: {device}")

# 加载模型
model = AutoModelForCausalLM.from_pretrained("gpt2")
tokenizer = AutoTokenizer.from_pretrained("gpt2")

# 移到 GPU
model = model.to(device)

# 生成
input_ids = tokenizer("Hello", return_tensors="pt").input_ids.to(device)
output = model.generate(input_ids, max_length=50)
print(tokenizer.decode(output[0]))
```

### 使用半精度（FP16）

```python
from transformers import AutoModelForCausalLM, AutoTokenizer
import torch

# 使用半精度加载
model = AutoModelForCausalLM.from_pretrained(
    "gpt2",
    torch_dtype=torch.float16,  # 半精度
    device_map="auto"            # 自动分配设备
)

tokenizer = AutoTokenizer.from_pretrained("gpt2")

# 生成
input_ids = tokenizer("Hello", return_tensors="pt").to(model.device)
output = model.generate(**input_ids, max_length=50)
print(tokenizer.decode(output[0]))
```

## 八、总结

### AutoClass 的关键点

| 概念 | 说明 |
|------|------|
| AutoClass | 自动选择正确的模型类 |
| AutoModelForCausalLM | 因果语言模型（GPT类） |
| AutoTokenizer | 分词器 |
| from_pretrained | 从 Hub 加载模型 |

### 使用流程

```
1. 选择模型名称
   比如 "gpt2"、"meta-llama/Llama-2-7b-hf"
   
2. 加载分词器
   tokenizer = AutoTokenizer.from_pretrained(model_name)
   
3. 加载模型
   model = AutoModelForCausalLM.from_pretrained(model_name)
   
4. 使用模型
   input_ids = tokenizer(text, return_tensors="pt")
   output = model.generate(**input_ids)
```
