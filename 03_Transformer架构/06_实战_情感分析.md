## 任务描述

**目标**：判断一段电影评论是正面（好评）还是负面（差评）。

**输入**：一段文字，比如 "This movie is amazing!"

**输出**：正面 或 负面

**示例**：

| 评论 | 情感 |
|------|------|
| "This movie is amazing!" | 正面 |
| "I hated this terrible film." | 负面 |
| "The story was boring." | 负面 |
| "A beautiful and touching story." | 正面 |

```python
# 转换成数字序列
["I", "love", "this", "movie"] → [1, 2, 3, 4]
```

### 第2步：搭建模型

```python
class SentimentClassifier(nn.Module):
    def __init__(self, vocab_size, d_model, num_heads, num_layers, d_ff, num_classes):
        super().__init__()
        
        # Transformer 编码器
        self.encoder = TransformerEncoder(vocab_size, d_model, num_heads, num_layers, d_ff)
        
        # 分类头
        self.classifier = nn.Linear(d_model, num_classes)
        
    def forward(self, x):
        # 1. 通过 Transformer 编码器
        encoded = self.encoder(x)  # (batch_size, seq_len, d_model)
        
        # 2. 池化：取所有位置的平均值
        pooled = encoded.mean(dim=1)  # (batch_size, d_model)
        
        # 3. 分类
        logits = self.classifier(pooled)  # (batch_size, num_classes)
        
        return logits
```

## 完整代码

```python
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import Dataset, DataLoader
import re
from collections import Counter

# ============ 数据准备 ============

def create_sample_data():
    """创建示例数据"""
    positive_texts = [
        "This movie is absolutely wonderful!",
        "Great acting and fantastic storyline.",
        "A masterpiece of cinema.",
        "I loved every moment of this film.",
        "Excellent performances from the cast.",
        "One of the best movies I've ever seen.",
        "The cinematography is breathtaking.",
        "A truly heartwarming story.",
        "Brilliant script and outstanding direction.",
        "I couldn't stop smiling throughout.",
    ]
    
    negative_texts = [
        "This movie was a complete waste of time.",
        "Terrible acting and a boring plot.",
        "I fell asleep halfway through.",
        "The worst film I've seen this year.",
        "Don't waste your money on this.",
        "The story makes no sense at all.",
        "Poor direction and awful script.",
        "I regret watching this movie.",
        "The characters are uninteresting.",
        "A disappointing experience.",
    ]
    
    texts = positive_texts + negative_texts
    labels = [1] * len(positive_texts) + [0] * len(negative_texts)
    
    return texts, labels

def build_vocab(texts):
    """构建词汇表"""
    word_counts = Counter()
    for text in texts:
        tokens = re.findall(r'\b\w+\b', text.lower())
        word_counts.update(tokens)
    
    vocab = {'<PAD>': 0, '<UNK>': 1}
    for idx, (word, _) in enumerate(word_counts.items(), start=2):
        vocab[word] = idx
    
    return vocab

class TextDataset(Dataset):
    def __init__(self, texts, labels, vocab, max_len=20):
        self.texts = texts
        self.labels = labels
        self.vocab = vocab
        self.max_len = max_len
    
    def __len__(self):
        return len(self.texts)
    
    def __getitem__(self, idx):
        text = self.texts[idx]
        label = self.labels[idx]
        
        tokens = re.findall(r'\b\w+\b', text.lower())
        indices = [self.vocab.get(t, self.vocab['<UNK>']) for t in tokens]
        
        # 填充或截断
        if len(indices) < self.max_len:
            indices += [0] * (self.max_len - len(indices))
        else:
            indices = indices[:self.max_len]
        
        return torch.tensor(indices, dtype=torch.long), torch.tensor(label, dtype=torch.long)

# ============ 模型定义 ============

class TransformerEncoder(nn.Module):
    """Transformer编码器（简化版）"""
    def __init__(self, vocab_size, d_model, num_heads, num_layers, d_ff, max_len=5000, dropout=0.1):
        super().__init__()
        
        self.embedding = nn.Embedding(vocab_size, d_model, padding_idx=0)
        
        # 位置编码
        pe = torch.zeros(max_len, d_model)
        position = torch.arange(0, max_len, dtype=torch.float).unsqueeze(1)
        div_term = torch.exp(torch.arange(0, d_model, 2).float() * (-torch.log(torch.tensor(10000.0)) / d_model))
        pe[:, 0::2] = torch.sin(position * div_term)
        pe[:, 1::2] = torch.cos(position * div_term)
        self.register_buffer('pe', pe.unsqueeze(0))
        
        # 使用 PyTorch 内置的 Transformer 编码器层
        encoder_layer = nn.TransformerEncoderLayer(
            d_model=d_model,
            nhead=num_heads,
            dim_feedforward=d_ff,
            dropout=dropout,
            batch_first=True
        )
        self.transformer = nn.TransformerEncoder(encoder_layer, num_layers=num_layers)
        
    def forward(self, x):
        x = self.embedding(x) * 0.1  # 缩放
        x = x + self.pe[:, :x.size(1), :]
        x = self.transformer(x)
        return x

class SentimentClassifier(nn.Module):
    """情感分类器"""
    def __init__(self, vocab_size, d_model, num_heads, num_layers, d_ff, num_classes):
        super().__init__()
        
        self.encoder = TransformerEncoder(vocab_size, d_model, num_heads, num_layers, d_ff)
        self.classifier = nn.Linear(d_model, num_classes)
        
    def forward(self, x):
        encoded = self.encoder(x)
        pooled = encoded.mean(dim=1)
        logits = self.classifier(pooled)
        return logits

# ============ 训练和测试 ============

def train_model():
    """训练模型"""
    # 准备数据
    texts, labels = create_sample_data()
    vocab = build_vocab(texts)
    
    dataset = TextDataset(texts, labels, vocab, max_len=20)
    dataloader = DataLoader(dataset, batch_size=4, shuffle=True)
    
    # 创建模型
    model = SentimentClassifier(
        vocab_size=len(vocab),
        d_model=32,
        num_heads=4,
        num_layers=2,
        d_ff=64,
        num_classes=2
    )
    
    # 训练
    criterion = nn.CrossEntropyLoss()
    optimizer = optim.Adam(model.parameters(), lr=0.001)
    
    for epoch in range(20):
        total_loss = 0
        correct = 0
        total = 0
        
        for texts_batch, labels_batch in dataloader:
            outputs = model(texts_batch)
            loss = criterion(outputs, labels_batch)
            
            optimizer.zero_grad()
            loss.backward()
            optimizer.step()
            
            total_loss += loss.item()
            predictions = torch.argmax(outputs, dim=1)
            correct += (predictions == labels_batch).sum().item()
            total += labels_batch.size(0)
        
        if (epoch + 1) % 5 == 0:
            accuracy = 100 * correct / total
            print(f"Epoch {epoch+1}: Loss={total_loss:.4f}, Accuracy={accuracy:.1f}%")
    
    return model, vocab

def test_model(model, vocab):
    """测试模型"""
    test_texts = [
        "This movie is amazing!",
        "I hated this terrible film.",
        "The story was boring.",
        "A beautiful and touching story.",
    ]
    
    print("\n测试结果：")
    for text in test_texts:
        tokens = re.findall(r'\b\w+\b', text.lower())
        indices = [vocab.get(t, vocab['<UNK>']) for t in tokens]
        
        # 填充到20
        if len(indices) < 20:
            indices += [0] * (20 - len(indices))
        else:
            indices = indices[:20]
        
        input_tensor = torch.tensor([indices], dtype=torch.long)
        
        with torch.no_grad():
            output = model(input_tensor)
            prediction = torch.argmax(output, dim=1).item()
            confidence = torch.softmax(output, dim=1)[0][prediction].item()
        
        sentiment = "正面" if prediction == 1 else "负面"
        print(f"  '{text}' → {sentiment} ({confidence:.1%})")

# ============ 主函数 ============

if __name__ == "__main__":
    print("训练 Transformer 情感分类器...\n")
    model, vocab = train_model()
    test_model(model, vocab)
```

## 七、从 Transformer 到语言模型

### 你可能有的疑问

> 学完了 Transformer，接下来学什么？为什么第4章突然冒出来一个"语言模型"？

### 答案：语言模型是 Transformer 的一个重要应用

在 3.2b 节，你学了 RNN 可以做**文本生成**：

```
输入："今天天气"
生成："今天天气真好"
```

**语言模型就是专门做这件事的模型！**

```
语言模型的任务：给定前面的词，预测下一个词

输入："今天天气"
模型预测："真" 的概率是 80%
输入："今天天气真"
模型预测："好" 的概率是 90%
...
```

### Transformer 和语言模型的关系

```
第3章：学了 Transformer（一种网络结构）
         ↓
第4章：用 Transformer 构建语言模型（一个应用）
         ↓
ChatGPT 就是一个用 Transformer 构建的语言模型！
```

**类比**：
- 第1章学了"神经网络"（结构）
- 第2章学了"卷积神经网络"（结构）
- 第2章最后做了"手写数字识别"（应用）

现在：
- 第3章学了"Transformer"（结构）
- 第4章学"语言模型"（应用）

### 语言模型为什么重要？

语言模型是 ChatGPT、Claude 等大语言模型的基础：

```
ChatGPT 的本质：
  输入："你好"

  ...
  
  最终生成："你好！有什么可以帮你的吗？"
```
