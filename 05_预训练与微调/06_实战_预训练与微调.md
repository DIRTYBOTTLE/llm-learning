## 二、微调预训练模型

### 准备微调数据

```python
# 微调数据：对话数据
train_data = [
    {
        "input": "你好",
        "output": "你好！有什么可以帮你的吗？"
    },
    {
        "input": "1+1等于几？",
        "output": "1+1等于2。"
    },
    {
        "input": "请介绍一下北京。",
        "output": "北京是中国的首都，也是中国的政治、文化中心。"
    },
    {
        "input": "今天天气怎么样？",
        "output": "我无法获取实时天气信息，建议你查看天气预报。"
    },
]
```

### 构造训练样本

```python
def prepare_training_data(data, tokenizer):
    """准备训练数据"""
    
    input_ids_list = []
    target_ids_list = []
    
    for sample in data:
        # 构造输入："用户：{问题} 助手："
        input_text = f"用户：{sample['input']} 助手："
        # 构造目标："{回答}"
        target_text = sample['output']
        
        # 编码
        input_ids = tokenizer.encode(input_text)
        target_ids = tokenizer.encode(target_text)
        
        # 合并（输入+目标）
        full_ids = input_ids + target_ids
        
        # 构造标签（只计算目标部分的损失）
        labels = [-100] * len(input_ids) + target_ids
        
        input_ids_list.append(full_ids)
        target_ids_list.append(labels)
    
    return input_ids_list, target_ids_list

# 准备数据
input_ids_list, target_ids_list = prepare_training_data(train_data, tokenizer)
```

### 微调训练循环

```python
from torch.optim import Adam

def finetune_model(model, train_data, num_epochs=3, lr=5e-5):
    """微调模型"""
    
    optimizer = Adam(model.parameters(), lr=lr)
    
    model.train()
    
    for epoch in range(num_epochs):
        total_loss = 0
        
        for input_ids, target_ids in train_data:
            # 转换成 tensor
            input_ids = torch.tensor([input_ids])
            target_ids = torch.tensor([target_ids])
            
            # 前向传播
            outputs = model(input_ids, labels=target_ids)
            loss = outputs.loss
            
            # 反向传播
            optimizer.zero_grad()
            loss.backward()
            optimizer.step()
            
            total_loss += loss.item()
        
        # 打印训练信息
        avg_loss = total_loss / len(train_data)
        print(f"Epoch {epoch+1}, Loss: {avg_loss:.4f}")
    
    return model

# 微调
model = finetune_model(model, list(zip(input_ids_list, target_ids_list)))
```

### 测试微调后的模型

```python
def test_finetuned_model(model, tokenizer, prompt):
    """测试微调后的模型"""
    
    # 构造输入
    input_text = f"用户：{prompt} 助手："
    inputs = tokenizer.encode(input_text, return_tensors="pt")
    
    # 生成
    outputs = model.generate(
        inputs,
        max_length=100,
        do_sample=True,
        temperature=0.7,
        pad_token_id=tokenizer.eos_token_id
    )
    
    # 解码输出
    text = tokenizer.decode(outputs[0], skip_special_tokens=True)
    
    # 提取助手的回答
    response = text.split("助手：")[-1].strip()
    
    return response

# 测试
prompts = ["你好", "1+1等于几？", "请介绍一下北京。"]

for prompt in prompts:
    response = test_finetuned_model(model, tokenizer, prompt)
    print(f"用户：{prompt}")
    print(f"助手：{response}")
    print()
```


### 使用 LoRA 进行高效微调

```python
from peft import LoraConfig, get_peft_model

# LoRA 配置
lora_config = LoraConfig(
    r=8,  # 低秩矩阵的秩
    lora_alpha=32,
    target_modules=["q_proj", "v_proj"],
    lora_dropout=0.05,
    bias="none",
    task_type="CAUSAL_LM"
)

# 应用 LoRA
model = get_peft_model(model, lora_config)

# 打印可训练参数量
model.print_trainable_parameters()
# 输出：trainable params: 4,194,304 || all params: 1,104,439,296 || trainable%: 0.38
```

### LoRA 的好处

```
全量微调：
  - 更新所有参数
  - 参数量：11亿
  - 显存需求：很大

LoRA 微调：
  - 只更新部分参数
  - 可训练参数：400万（0.38%）
  - 显存需求：小很多
  
LoRA 让个人电脑也能微调大模型！
```

## 练习答案

### 练习1答案

```python
from transformers import GPT2LMHeadModel, GPT2Tokenizer

tokenizer = GPT2Tokenizer.from_pretrained("gpt2")
model = GPT2LMHeadModel.from_pretrained("gpt2")

inputs = tokenizer.encode("今天天气", return_tensors="pt")
outputs = model.generate(inputs, max_length=50)
print(tokenizer.decode(outputs[0]))
```

### 练习2答案

1. **数据格式**：输入（用户问题）+ 目标（期望回答）
2. **为什么只计算目标部分**：因为模型需要学习的是"怎么回答"，不是"怎么理解问题"
3. **LoRA**：低秩适应，只更新部分参数，好处是显存需求小，个人电脑也能微调大模型
